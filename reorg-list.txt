function onCharBlur() {
  setTimeout(() => commitChars(true), 160)

/* el-autocomplete 的联想源：从「尚未选中的角色」里按输入过滤 */
function fetchCharSuggestions(queryString, cb) {
  const q = (queryString || '').toLowerCase()
  const list = q
    ? charSuggestions.value.filter((n) => n.toLowerCase().includes(q))
    : charSuggestions.value
  cb(list.map((value) => ({ value })))
}

## 2026 年 11 月

### 2026-11-18 · 旅闻编辑器删除入口修正 · 三页危险按钮高度统一

- **起因**：① 手机端点进某条旅闻、再点「返回列表」后，底部「删除此条」仍留在列表页——返回只切了面板，选中项没清，`v-if="entry"` 依然成立。② 三个页面的底部危险按钮高度不统一：编辑器「删除此条」是默认 32px，游逸旅闻「清空全部记录」与赠礼套装「重置全部数据」是 `size="small"`（24px）。
- **经过**：编辑器底部危险区条件改为 `entry && mobilePane === 'form'`，与「返回列表」一样只在表单栏出现；游逸旅闻与赠礼套装的底部按钮去掉 `size="small"`，与编辑器统一为默认高度。同批按各自格式更新 `AGENTS.md`（最后合并行）与 `CHANGELOG.md`（当日条目）。
- **结果**：手机端进入编辑表单才出现删除入口，返回列表即消失；三个页面的底部危险按钮同为默认高度，视觉不再跳档。

## 2026 年 11 月

### 2026-11-18 · 尘歌壶字典编辑器迁 EP 完成 · 四页原生标签清零

- **起因**：尘歌壶字典编辑器还在用自造控件（`Btn` / `Modal` / `Dropdown` / `Combobox` / `Seg` / `Field` / `Icon`），与已完成迁移的旅闻编辑器不是一套；全站四页还留着原生 `button` / `input` 标签。
- **经过**：编辑器控件全量换 EP —— 工具条与列表脚改 `el-button`，搜索框改 `el-input`（clearable），两个筛选改 `el-select`，地区联想改 `el-select`（允许自建），偏好角色添加器改 `el-autocomplete`（联想源取「未选角色」），室内/室外改 `el-segmented`，图片 URL 与 id 改 `el-input`，四个弹窗改 `el-dialog`（确认后手动关闭），图标改 `@element-plus/icons-vue`；底部「删除此套」改为仅在表单栏出现（与旅闻编辑器同一处手机端 bug）。GiftSet 页两张套装卡片头、GiftSet 编辑器列表项与角色移除格改 `role="button"` 的 `div`（保留键盘激活），三处隐藏文件输入补 `aria-hidden` 与 `tabindex`。
- **结果**：四个页面自造控件与原生交互标签清零，控件尺寸、焦点环、弹窗行为全站统一走 EP 默认，后续不再有两套样式并存。
