# 清单文件：给 .github/workflows/reorg.yml 读的。四段，每段开头一行 == 段名 ==。
#   MOVES    一行一条：旧路径 -> 新路径
#   EDITS    一条一个块：### 文件路径，然后 <<<< 原文 >>>>，再 <<<< 新文 >>>>（原文逐字、在该文件里唯一）
#            —— 这一段里不要写 # 注释行（会被当成文件路径）
#            —— 一个 ### 只能跟一对块；同一文件要改多处，就重复写几遍 ### 路径
#   DELETES  一行一个路径（文件或目录都行）
#   CHECKS   一行一条命令，非 0 即中止
# 跑法：改这个文件 → 提交 → Actions → Reorg → Run workflow（清单名 = 分支名）
#
# 这批：boot-guard —— 修「页面未能加载」误报。
#   根因：首屏要下约 1 MB 脚本，守卫 4 秒就判死；国内访问 GitHub Pages 的 CDN 又常被限速。
#   改法：4 秒定时改成「每 500ms 看 #app，挂载即停、12 秒才报错」；文案改成网络优先；
#         加一个能点的「重新加载」按钮。

        /* 首屏要下约 1 MB 脚本，4 秒在慢网下根本不够，会把「慢」误报成「失败」。
           改成每 500ms 看一眼 #app：挂载了就立刻停，12 秒还没内容才报错 */
        var waited = 0;
        var watch = setInterval(function () {
          waited += 500;
          var el = document.getElementById("app");
          if (el && el.children.length > 0 && !el.dataset.fatal) {
            clearInterval(watch);
            return;
          }
          if (waited >= 12000) {
            clearInterval(watch);
            recover();
          }
        }, 500);
